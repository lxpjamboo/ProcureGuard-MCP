        // Render function to populate screen
        function renderTenderView(key) {
            const data = TENDER_DATABASE[key];

            // Render Header KPIs
            document.getElementById('tenderBudget').innerText = data.budget;
            document.getElementById('tenderBidsCount').innerText = data.bidsCount;
            document.getElementById('kpiRiskScore').innerText = data.riskScore;
            document.getElementById('kpiDrift').innerText = data.maxDrift;
            document.getElementById('kpiSpecMatch').innerText = data.specMatched;

            // Risk KPI Styling
            const kpiCard = document.getElementById('kpiRiskCard');
            const kpiBadge = document.getElementById('kpiRiskBadge');
            const kpiScore = document.getElementById('kpiRiskScore');
            
            if(data.riskScore.startsWith('8') || data.riskScore.startsWith('9')) {
                kpiCard.className = "bg-panel border border-signalRed/50 rounded-lg p-3.5 flex flex-col justify-between transition-all duration-300 red-flag-glow";
                kpiBadge.className = "text-[10px] font-mono font-bold px-1.5 py-0.5 rounded bg-signalRed/20 text-signalRed border border-signalRed/50";
                kpiBadge.innerText = "HIGH RISK";
                kpiScore.className = "text-2xl font-mono font-bold text-signalRed";
            } else if(data.riskScore.startsWith('4')) {
                kpiCard.className = "bg-panel border border-phosphorGold/50 rounded-lg p-3.5 flex flex-col justify-between transition-all duration-300 gold-flag-glow";
                kpiBadge.className = "text-[10px] font-mono font-bold px-1.5 py-0.5 rounded bg-phosphorGold/20 text-phosphorGold border border-phosphorGold/50";
                kpiBadge.innerText = "MODERATE";
                kpiScore.className = "text-2xl font-mono font-bold text-phosphorGold";
            } else {
                kpiCard.className = "bg-panel border border-emeraldVerified/50 rounded-lg p-3.5 flex flex-col justify-between transition-all duration-300 emerald-glow";
                kpiBadge.className = "text-[10px] font-mono font-bold px-1.5 py-0.5 rounded bg-emeraldVerified/20 text-emeraldVerified border border-emeraldVerified/50";
                kpiBadge.innerText = "VERIFIED";
                kpiScore.className = "text-2xl font-mono font-bold text-emeraldVerified";
            }

            // Render Bidders
            const bidderContainer = document.getElementById('bidderCardsContainer');
            bidderContainer.innerHTML = '';
            data.bidders.forEach(b => {
                const colorClass = b.badgeColor === 'signalRed' ? 'border-signalRed/40 text-signalRed bg-signalRed/10' :

            });

            // Render MCP Console Terminal Logs

                    <div class="space-y-0.5">

                        <p class="text-xs text-slate-300 leading-relaxed font-sans">${rf.desc}</p>
                    </div>
                `;
            });

            lucide.createIcons();
        }

        function triggerMcpEvaluation() {
            const btn = document.getElementById('runPipelineBtn');
            const terminal = document.getElementById('mcpTerminal');

            btn.disabled = true;
            btn.classList.add('opacity-50');
            btn.innerHTML = `<i data-lucide="loader-2" class="w-4 h-4 animate-spin"></i><span>Invoking MCP Server...</span>`;
            lucide.createIcons();

            terminal.innerHTML += `
                <div class="text-sovereignCyan font-bold pt-2 border-t border-stark animate-pulse">
                    >>> INITIATING REAL-TIME MCP FORENSIC RE-SCAN...
                </div>
            `;
            terminal.scrollTop = terminal.scrollHeight;

            setTimeout(() => {
                renderTenderView(currentTenderKey);
                btn.disabled = false;
                btn.classList.remove('opacity-50');
                btn.innerHTML = `<i data-lucide="play" class="w-4 h-4 fill-black"></i><span class="font-mono uppercase tracking-wider">Execute MCP Pipeline</span>`;
                lucide.createIcons();
            }, 1200);
        }

        // Human-in-the-Loop Signoff Commitment function
        function commitAuditLog() {
            const officerId = document.getElementById('officerIdInput').value.trim();
            const notes = document.getElementById('officerNotes').value.trim();

            if (!officerId) {
                alert("OFFICER REJECTION: Officer ID / Badge No is required for legal audit compliance.");
                return;
            }
            if (!notes) {
                alert("GOVERNANCE REQUIREMENT: You must provide mandatory written justification notes before committing the sign-off.");
                return;
            }

            // Lock controls
            document.getElementById('signoffControls').classList.add('opacity-40', 'pointer-events-none');
            document.getElementById('committedBanner').classList.remove('hidden');

            // Generate pseudo random cryptographic hash
            const hash = '0x' + Array.from({length: 16}, () => Math.floor(Math.random()*16).toString(16)).join('');
            document.getElementById('committedHash').innerText = hash;

            // Update HITL status KPI badge
            const kpiHitlStatus = document.getElementById('kpiHitlStatus');
            kpiHitlStatus.className = "text-sm font-mono font-bold text-emeraldVerified flex items-center gap-1";
            kpiHitlStatus.innerHTML = `<i data-lucide="check-circle" class="w-4 h-4"></i> SEALED & SIGNED`;

            const hitlSection = document.getElementById('hitlSection');
            hitlSection.className = "bg-panel border-2 border-emeraldVerified/60 rounded-lg p-5 emerald-glow relative overflow-hidden transition-all duration-300";

            const lockIcon = document.getElementById('lockIconContainer');
            lockIcon.className = "w-10 h-10 rounded bg-emeraldVerified/10 border border-emeraldVerified flex items-center justify-center text-emeraldVerified";
            lockIcon.innerHTML = `<i data-lucide="file-check" class="w-5 h-5"></i>`;

            lucide.createIcons();
        }
